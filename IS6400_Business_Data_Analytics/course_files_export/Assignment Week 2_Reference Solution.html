import matplotlib.pyplot as plt
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline

alphas = [0.001, 0.01, 0.1]

# Set up a figure with 3 subplots (1 row, 3 columns)
fig, axes = plt.subplots(1, 3, figsize=(18, 5))

for ax, alpha in zip(axes, alphas):
    # Create pipeline with StandardScaler to fix the LinAlgWarning
    model = make_pipeline(StandardScaler(), Ridge(alpha=alpha))
    
    # Fit the model
    model.fit(data[predictors], data['y'])
    
    # Predict
    y_pred = model.predict(data[predictors])
    
    # Plot
    ax.plot(data['x'], y_pred, linewidth=4, label='Ridge Prediction')
    ax.plot(data['x'], data['y'], '.', markersize=10, label='Actual Data')
    ax.set_title(f'Alpha = {alpha}', fontsize=16)
    ax.set_xlabel('x')
    ax.set_ylabel('y')
    ax.legend()

plt.tight_layout()
plt.show()

import pandas as pd
import numpy as np
from sklearn.linear_model import Ridge
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error

# Define the alpha values
alphas = [0.001, 0.01, 0.1]

# Create an empty list to store the results
results_list = []

# Loop through each alpha
for alpha in alphas:
    # Use a pipeline to standardize the data first, then apply Ridge
    # This fixes the Ill-conditioned matrix warning
    ridge = make_pipeline(StandardScaler(), Ridge(alpha=alpha))
    
    # Fit the model
    ridge.fit(data[predictors], data['y'])
    
    # Predict
    y_pred = ridge.predict(data[predictors])
    
    # Calculate all three metrics
    r2 = r2_score(data['y'], y_pred)
    mae = mean_absolute_error(data['y'], y_pred)
    rmse = np.sqrt(mean_squared_error(data['y'], y_pred)) # Correct RMSE formula
    
    # Append the results as a dictionary to the list
    results_list.append({
        'Alpha': alpha,
        'R² Score': r2,
        'MAE': mae,
        'RMSE': rmse
    })

# Convert the list of dictionaries into a pandas DataFrame
results_df = pd.DataFrame(results_list)

# Display the comparison table
print("Comparison Table for Ridge Regression:")
display(results_df) # Use display() if in Jupyter Notebook, otherwise use print()
